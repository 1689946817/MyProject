import os
import json
import csv
import torch
from swift.infer_engine import TransformersEngine, RequestConfig, InferRequest
from swift import get_model_processor, get_template
from peft import PeftModel
from tqdm import tqdm

# ================= 配置区 =================
MODEL_ID_OR_PATH = '/mnt/workspace/Model/Qwen/Qwen3.5-4B'
CKPT_75 = '/mnt/workspace/output_unidoc_qwen4b/v8-20260330-002448/checkpoint-75'
CKPT_140 = '/mnt/workspace/output_unidoc_qwen4b/v8-20260330-002448/checkpoint-140'
VAL_JSONL = 'finetune/data/val.jsonl'
OUTPUT_BASE = 'val_eval_results'

# 核心系统提示词
SYSTEM_PROMPT = """你是一名多模态数据分析专家。请解析给定图像，并按照以下逻辑生成描述文本。要求：不要输出任何类似 "属性名:" 或 "JSON标签" 的固定格式，统一使用 [###] 作为不同部分的唯一分隔符。生成逻辑如下：[第一部分：定义分类] 开头必须明确指出该图片的类型（这部分可以稍微详细些，例如：它是某某机器的操作流程图、历史地图、职场截图、自然风景、工程逻辑图、手绘草图等）。[第二部分：语义摘要] 紧接分类后，用一段话概括图像的核心意图和功能。[第三部分：视觉细节] 描述主体、空间关系、颜色及环境。[第四部分：文字提取] 罗列图中所有的关键文字信息。[第五部分：检索关键词] 提供 10 个有助于精准检索的词汇。示例格式：[###] 该图片是一份职场维权指南截图。主要内容是关于“裁员黄金一小时”应对策略..."""
# ==========================================

def get_val_data(jsonl_path):
    data_list = []
    if not os.path.exists(jsonl_path):
        return data_list
    with open(jsonl_path, 'r', encoding='utf-8') as f:
        for line in f:
            data = json.loads(line)
            img_path = data['images'][0]
            if not img_path.startswith('/'):
                img_path = os.path.join('/mnt/workspace', img_path)
            gt_text = data['messages'][1]['content'] if len(data['messages']) > 1 else ""
            data_list.append({"image": img_path, "gt": gt_text})
    return data_list

def run_official_engine_inference(model_id, lora_path, data_list, label):
    print(f"\n>>> [Official Engine] 正在初始化 {label}...")
    
    # 1. 加载模型和处理器
    model, tokenizer = get_model_processor(model_id)
    
    # 2. 如果有 LoRA，则加载适配器
    if lora_path is not None:
        model = PeftModel.from_pretrained(model, lora_path)
    
    # 3. 设置对话模板
    template_type = model.model_meta.template
    template = get_template(tokenizer, template_type=template_type, default_system=SYSTEM_PROMPT)
    
    # 4. 初始化推理引擎 (这里 max_batch_size 设为 1 以保证显存安全)
    engine = TransformersEngine(model, template=template, max_batch_size=1)
    request_config = RequestConfig(max_tokens=2048, temperature=0)
    
    results = []
    for item in tqdm(data_list, desc=f"推理中({label})"):
        # 遵循官方消息格式，使用 <image> 标签
        infer_request = InferRequest(
            messages=[{'role': 'user', 'content': f'<image>请解析这张图像。'}],
            images=[item['image']]
        )
        resp_list = engine.infer([infer_request], request_config)
        results.append(resp_list[0].choices[0].message.content)
    
    # 显存清理：非常关键，防止切换模型时 OOM
    del engine
    del model
    torch.cuda.empty_cache()
    return results

def main():
    val_data = get_val_data(VAL_JSONL)
    if not val_data:
        print(f"错误：无法读取验证集数据，请检查路径: {VAL_JSONL}")
        return

    # 依次运行三个模型
    base_out = run_official_engine_inference(MODEL_ID_OR_PATH, None, val_data, "原始模型")
    ckpt75_out = run_official_engine_inference(MODEL_ID_OR_PATH, CKPT_75, val_data, "Ckpt-75 (Best)")
    ckpt140_out = run_official_engine_inference(MODEL_ID_OR_PATH, CKPT_140, val_data, "Ckpt-140 (Final)")
    
    final_results = []
    for i, item in enumerate(val_data):
        final_results.append({
            "id": i + 1,
            "image": item['image'],
            "ground_truth": item['gt'],
            "res_base": base_out[i],
            "res_75": ckpt75_out[i],
            "res_140": ckpt140_out[i]
        })
    
    # 保存 1: JSONL (打分必备)
    with open(f"{OUTPUT_BASE}.jsonl", 'w', encoding='utf-8') as f:
        for entry in final_results:
            f.write(json.dumps(entry, ensure_ascii=False) + '\n')
    
    # 保存 2: CSV (统计必备)
    with open(f"{OUTPUT_BASE}.csv", 'w', encoding='utf-8-sig', newline='') as f:
        writer = csv.writer(f)
        writer.writerow(["ID", "ImagePath", "GT", "Base", "Ckpt75", "Ckpt140"])
        for r in final_results:
            writer.writerow([r['id'], r['image'], r['ground_truth'], r['res_base'], r['res_75'], r['res_140']])
    
    # 保存 3: MD (展示必备)
    with open(f"{OUTPUT_BASE}.md", 'w', encoding='utf-8') as f:
        f.write(f"# UniDoc-Bench 实验对比报告 - 张琪 (东北大学)\n\n")
        for r in final_results:
            f.write(f"## 样本 {r['id']}\n**路径**: `{r['image']}`\n")
            f.write(f"### [1. 原始 Qwen3.5-4B]\n{r['res_base']}\n\n")
            f.write(f"### [2. 微调 Ckpt-75 (Best)]\n{r['res_75']}\n\n")
            f.write(f"### [3. 微调 Ckpt-140 (Final)]\n{r['res_140']}\n\n---\n")
    
    print(f"\n✅ 官方架构推理完成！产出文件：{OUTPUT_BASE}.jsonl/csv/md")

if __name__ == '__main__':
    main()
