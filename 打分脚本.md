import json
import csv
import base64
import os
from openai import OpenAI
from tqdm import tqdm

# ================= 配置区 =================
client = OpenAI(
    api_key="你的_GPT4o_API_KEY", 
    base_url="https://api.openai.com/v1" # 或者你使用的中转地址
)

INPUT_JSONL = 'val_eval_results.jsonl'
OUTPUT_CSV = 'gpt4o_vision_scores.csv'
JUDGE_MODEL = "gpt-4o" # 确保使用的是支持视觉的模型

# 训练时使用的系统提示词（法官需要据此判断模型是否“听话”）
ORIGINAL_SYSTEM_PROMPT = """你是一名多模态数据分析专家。请解析给定图像，并按照以下逻辑生成描述文本。要求：不要输出任何类似 "属性名:" 或 "JSON标签" 的固定格式，统一使用 [###] 作为不同部分的唯一分隔符。生成逻辑如下：[第一部分：定义分类] 开头必须明确指出该图片的类型...[第二部分：语义摘要]...[第三部分：视觉细节]...[第四部分：文字提取]...[第五部分：检索关键词]..."""

# 裁判的评分标准（Prompt）
JUDGE_INSTRUCTIONS = f"""你是一名严谨的视觉语言模型评测专家。
现在我会给你一张【原始图片】，以及三个模型针对这张图片生成的回答。
这三个模型在推理时都接收到了相同的【系统提示词要求】。

【系统提示词要求】：
{ORIGINAL_SYSTEM_PROMPT}

请从以下维度对三个模型的回答进行打分（每个维度 1-10 分）：
1. 格式遵循度 (Format): 是否严格使用 [###] 分隔五个部分，无冗余标签，逻辑清晰。
2. OCR 准确性 (OCR): 提取的文字是否与图片内容完全一致，无幻觉，无错漏。
3. 描述深度 (Semantics): 对图片分类、意图和视觉细节的分析是否准确且深刻。

请直接返回 JSON 格式结果，不要有任何多余文字。格式如下：
{{
  "score_base": {{"format": 8, "ocr": 5, "semantics": 7}},
  "score_75": {{"format": 10, "ocr": 9, "semantics": 9}},
  "score_140": {{"format": 9, "ocr": 8, "semantics": 8}},
  "reason": "简短的综合评价"
}}
"""
# ==========================================

def encode_image(image_path):
    """将本地图片转为 Base64 编码"""
    with open(image_path, "rb") as image_file:
        return base64.b64encode(image_file.read()).decode('utf-8')

def judge_with_gpt4o_vision(item):
    """调用 GPT-4o 视觉能力进行打分"""
    base64_image = encode_image(item['image'])
    
    user_content = [
        {
            "type": "text",
            "text": f"参考真值(Ground Truth): {item['ground_truth']}\n\n"
                    f"待评测模型回答：\n"
                    f"--- 模型 A (原始): {item['res_base']}\n"
                    f"--- 模型 B (Ckpt-75): {item['res_75']}\n"
                    f"--- 模型 C (Ckpt-140): {item['res_140']}"
        },
        {
            "type": "image_url",
            "image_url": {
                "url": f"data:image/png;base64,{base64_image}"
            }
        }
    ]
    
    response = client.chat.completions.create(
        model=JUDGE_MODEL,
        messages=[
            {"role": "system", "content": JUDGE_INSTRUCTIONS},
            {"role": "user", "content": user_content}
        ],
        response_format={ "type": "json_object" },
        max_tokens=1000
    )
    return json.loads(response.choices[0].message.content)

def main():
    if not os.path.exists(INPUT_JSONL):
        print(f"错误：找不到输入文件 {INPUT_JSONL}")
        return

    scored_results = []
    with open(INPUT_JSONL, 'r', encoding='utf-8') as f:
        lines = f.readlines()
    
    print(f"🚀 开始 GPT-4o 视觉自动化评估，共 {len(lines)} 个样本...")
    
    for line in tqdm(lines):
        item = json.loads(line)
        try:
            res = judge_with_gpt4o_vision(item)
            
            # 计算综合平均分
            avg_base = sum(res['score_base'].values()) / 3
            avg_75 = sum(res['score_75'].values()) / 3
            avg_140 = sum(res['score_140'].values()) / 3
            
            scored_results.append([
                item['id'], 
                f"{avg_base:.2f}", f"{avg_75:.2f}", f"{avg_140:.2f}",
                json.dumps(res['score_75'], ensure_ascii=False), # 记录 Best 模型的详细得分
                res['reason']
            ])
        except Exception as e:
            print(f"样本 {item['id']} 评估出错: {e}")
    
    # 保存结果为 CSV
    with open(OUTPUT_CSV, 'w', encoding='utf-8-sig', newline='') as f:
        writer = csv.writer(f)
        writer.writerow(["ID", "Base_Avg", "Ckpt75_Avg", "Ckpt140_Avg", "Ckpt75_Detail", "Reason"])
        writer.writerows(scored_results)
    
    print(f"\n✅ 评估完成！结果已保存至: {OUTPUT_CSV}")

if __name__ == '__main__':
    main()
