  从 COCO 生成评估 JSON：

  COCO val2017 有 5000 张图，建议取 100-200 张子集。以下是生成脚本（保存为
  scripts/prepare_coco_eval.py）：

  """
  从 COCO val2017 annotations 生成评估子集 JSON。
  每条记录包含：query（caption）、relevant_ids（图片ID）、reference_answer。
  """
  import json
  import random
  from pathlib import Path

  # 路径配置
  CAPTIONS_FILE = "data/annotations/captions_val2017.json"
  OUTPUT_FILE = "data/coco_subset_eval.json"
  SUBSET_SIZE = 100  # 取多少条查询
  RANDOM_SEED = 42

  random.seed(RANDOM_SEED)

  with open(CAPTIONS_FILE, encoding="utf-8") as f:
      coco = json.load(f)

  # 建立 image_id -> file_name 映射
  id_to_filename = {img["id"]: img["file_name"] for img in coco["images"]}

  # 每张图取第一条 caption 作为 query
  seen_image_ids = set()
  samples = []

  for ann in coco["annotations"]:
      img_id = ann["image_id"]
      if img_id in seen_image_ids:
          continue
      seen_image_ids.add(img_id)

      # COCO image_id 转为 12 位零填充字符串（与文件名一致）
      img_id_str = str(img_id).zfill(12)
      caption = ann["caption"].strip()

      samples.append({
          "query": caption,
          "relevant_ids": [img_id_str],
          "reference_answer": caption,  # 用 caption 本身作为参考答案
          "file_name": id_to_filename[img_id]
      })

  # 随机抽取子集
  samples = random.sample(samples, min(SUBSET_SIZE, len(samples)))

  with open(OUTPUT_FILE, "w", encoding="utf-8") as f:
      json.dump(samples, f, ensure_ascii=False, indent=2)

  print(f"Generated {len(samples)} samples -> {OUTPUT_FILE}")

  运行：
  python scripts/prepare_coco_eval.py

  1.3 使用自定义领域数据集

  如果不用 COCO，自定义数据集的 JSON 格式如下：

  [
    {
      "query": "一张红色跑车停在路边的图片",
      "relevant_ids": ["img_001", "img_002"],
      "reference_answer": "图中是一辆红色跑车停在路边，背景是城市街道。"
    },
    {
      "query": "两个人在海边散步",
      "relevant_ids": ["img_003"],
      "reference_answer": "图中有两个人在海边沙滩上散步，远处是大海。"
    }
  ]

  字段说明：
  - query：检索查询文本（自然语言描述）
  - relevant_ids：该查询对应的正确图片 ID 列表（与系统中 ImageRecord.id 对齐）
  - reference_answer：参考答案，用于生成质量评估的 Answer Correctness 指标

  关键：relevant_ids 必须与系统中的图片 ID 完全一致。 图片上传后，ID 是
  UUID，需要从数据库中查询对应关系。

  # scripts/batch_upload.py
  """批量上传图片到知识库，并记录 ID 映射关系。"""
  import requests
  import json
  from pathlib import Path

  BACKEND_URL = "http://localhost:9090"
  IMAGE_DIR = Path("data/coco_images")
  ID_MAP_FILE = "data/image_id_map.json"  # 保存 filename -> uuid 映射

  id_map = {}
  image_files = list(IMAGE_DIR.glob("*.jpg")) + list(IMAGE_DIR.glob("*.png"))

  for i, img_path in enumerate(image_files):
      print(f"[{i+1}/{len(image_files)}] Uploading {img_path.name}...")
      with open(img_path, "rb") as f:
          resp = requests.post(
              f"{BACKEND_URL}/api/knowledge-base/upload",
              files={"file": (img_path.name, f, "image/jpeg")}
          )
      if resp.status_code == 200:
          data = resp.json()
          uuid = data.get("id") or data.get("image_id")
          id_map[img_path.name] = uuid
          print(f"  -> ID: {uuid}")
      else:
          print(f"  -> FAILED: {resp.text}")

  with open(ID_MAP_FILE, "w") as f:
      json.dump(id_map, f, indent=2)
  print(f"Saved ID map to {ID_MAP_FILE}")

  # scripts/update_relevant_ids.py
  import json

  with open("data/image_id_map.json") as f:
      id_map = json.load(f)  # filename -> uuid

  with open("data/coco_subset_eval.json") as f:
      samples = json.load(f)

  for sample in samples:
      new_ids = []
      for item in sample["relevant_ids"]:
          # item 可能是 "000000123456"，对应文件名 "000000123456.jpg"
          filename = item + ".jpg" if not item.endswith(".jpg") else item
          uuid = id_map.get(filename)
          if uuid:
              new_ids.append(uuid)
          else:
              print(f"[WARN] No UUID found for {filename}")
      sample["relevant_ids"] = new_ids

  with open("data/coco_subset_eval.json", "w", encoding="utf-8") as f:
      json.dump(samples, f, ensure_ascii=False, indent=2)
  print("Updated relevant_ids in coco_subset_eval.json")

  4.3 批量运行并自动保存结果（推荐）

  # scripts/run_all_retrieval_eval.py
  """批量运行所有方法的检索评估，结果保存到 JSON。"""
  import subprocess
  import json
  import re
  from pathlib import Path

  METHODS = ["proposed", "baseline_text", "baseline_clip", "baseline_ocr"]
  DATASET = "data/coco_subset_eval.json"
  TOP_K = 10
  RESULTS_FILE = "data/retrieval_results.json"

  all_results = {}

  for method in METHODS:
      print(f"\n{'='*50}")
      print(f"Evaluating: {method}")
      print('='*50)

      result = subprocess.run(
          ["python", "-m", "evaluation.run_offline_eval",
           "--dataset-path", DATASET,
           "--method", method,
           "--top-k", str(TOP_K)],
          capture_output=True, text=True
      )

      output = result.stdout
      print(output)

      # 解析输出
      metrics = {}
      for line in output.splitlines():
          m = re.match(r"(Recall@\d+|MRR|mAP@\d+): ([0-9.]+)", line)
          if m:
              metrics[m.group(1)] = float(m.group(2))

      all_results[method] = metrics

  with open(RESULTS_FILE, "w") as f:
      json.dump(all_results, f, indent=2)

  print(f"\nAll results saved to {RESULTS_FILE}")

  # scripts/plot_generation_results.py
  import json
  import pandas as pd
  import matplotlib.pyplot as plt
  import matplotlib
  import numpy as np

  matplotlib.rcParams['font.family'] = 'SimHei'
  matplotlib.rcParams['axes.unicode_minus'] = False

  methods = {
      "proposed": "MLLM描述桥接",
      "baseline_text": "纯文本RAG",
      "baseline_clip": "跨模态嵌入",
      "baseline_ocr": "OCR文本",
  }

  metrics_to_plot = [
      "Answer Correctness",
      "Answer Relevancy",
      "Image Faithfulness",
      "Image Context Relevancy",
  ]

  all_scores = {}
  for method_key in methods:
      path = f"data/rag_outputs/eval_result_{method_key}.json"
      df = pd.read_json(path)
      scores = {}
      for m in metrics_to_plot:

              scores[m] = df[col].mean()
          else:
              scores[m] = 0.0

  fig, ax = plt.subplots(figsize=(8, 8), subplot_kw=dict(polar=True))

  N = len(metrics_to_plot)

  colors = ["#e74c3c", "#3498db", "#2ecc71", "#f39c12"]

  for (method_key, label), color in zip(methods.items(), colors):
      vals = [all_scores[method_key][m] for m in metrics_to_plot]
      vals += vals[:1]
      ax.plot(angles, vals, color=color, linewidth=2, label=label)
      ax.fill(angles, vals, color=color, alpha=0.1)

  ax.set_xticks(angles[:-1])
  ax.set_xticklabels(metrics_to_plot, fontsize=10)
  ax.set_ylim(0, 1)
  ax.set_title("RAG 生成质量多维度对比", fontsize=14, pad=20)
  ax.legend(loc="upper right", bbox_to_anchor=(1.3, 1.1))

  ---
  注意事项

  关于 relevant_ids 对齐：这是最容易出错的地方。评估 JSON 里的 ID 必须和 ChromaDB 里存的 ID
   完全一致（UUID 格式）。建议上传完图片后立即运行 update_relevant_ids.py。

  关于 OCR 基线：COCO 是自然场景图，大多数图片几乎没有文字，OCR 基线预期表现很差（Recall@10
   可能低于 0.1），这正是论文中用来衬托 MLLM 描述方法优势的对照组。

  关于 baseline_text 的"上界"定位：它使用人工标注的 COCO
  caption，在实际部署中不可用，因此即使它的指标高于 proposed 方法，也不影响 proposed       
  方法的实用价值。

  关于 API 调用成本：100 条查询 × 4 种方法 × top-5 图片，生成阶段约需调用 MLLM 400
  次，评估阶段如用 GPT-4V 还需额外 400 次调用，注意控制成本。
